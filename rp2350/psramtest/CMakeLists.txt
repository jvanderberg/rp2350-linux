cmake_minimum_required(VERSION 3.13)
set(PICO_PLATFORM rp2350-arm-s)
set(PICO_BOARD pico2)
include(pico_sdk_import.cmake)
project(psramtest C CXX ASM)
pico_sdk_init()
add_executable(psramtest main.c)
target_link_libraries(psramtest pico_stdlib hardware_clocks hardware_vreg)
pico_enable_stdio_rtt(psramtest 1)
pico_enable_stdio_uart(psramtest 0)
pico_enable_stdio_usb(psramtest 0)
pico_add_extra_outputs(psramtest)

# Overclocked variant: 300 MHz at 1.20 V, 5 memtest rounds.
add_executable(psramtest300 main.c)
target_compile_definitions(psramtest300 PRIVATE SYS_KHZ=300000 PASSES=5)
target_link_libraries(psramtest300 pico_stdlib hardware_clocks hardware_vreg)
pico_enable_stdio_rtt(psramtest300 1)
pico_enable_stdio_uart(psramtest300 0)
pico_enable_stdio_usb(psramtest300 0)
pico_add_extra_outputs(psramtest300)

# ~350 MHz variants (348 = 12 MHz x 87 / 3; exact 350 is not reachable) at 1.20 V and 1.30 V.
foreach(mv 1200 1300)
  if(mv EQUAL 1200)
    set(vr VREG_VOLTAGE_1_20)
  else()
    set(vr VREG_VOLTAGE_1_30)
  endif()
  add_executable(psramtest348_${mv} main.c)
  target_compile_definitions(psramtest348_${mv} PRIVATE SYS_KHZ=348000 PASSES=5 VREG=${vr} VREG_MV=${mv})
  target_link_libraries(psramtest348_${mv} pico_stdlib hardware_clocks hardware_vreg)
  pico_enable_stdio_rtt(psramtest348_${mv} 1)
  pico_enable_stdio_uart(psramtest348_${mv} 0)
  pico_enable_stdio_usb(psramtest348_${mv} 0)
  pico_add_extra_outputs(psramtest348_${mv})
endforeach()

# 348 MHz with PSRAM divider 3 (116 MHz SCK, above the 109 MHz rating).
add_executable(psramtest348_div3 main.c)
target_compile_definitions(psramtest348_div3 PRIVATE SYS_KHZ=348000 PASSES=5 PSRAM_CLKDIV=3)
target_link_libraries(psramtest348_div3 pico_stdlib hardware_clocks hardware_vreg)
pico_enable_stdio_rtt(psramtest348_div3 1)
pico_enable_stdio_uart(psramtest348_div3 0)
pico_enable_stdio_usb(psramtest348_div3 0)
pico_add_extra_outputs(psramtest348_div3)
